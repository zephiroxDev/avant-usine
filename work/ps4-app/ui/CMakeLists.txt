cmake_minimum_required(VERSION 3.16)
project(avant_usine_native_ui LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(ENGINE ${CMAKE_CURRENT_SOURCE_DIR}/../reference/lvgl)
if(NOT EXISTS ${ENGINE}/lvgl.h)
  message(FATAL_ERROR "Run scripts/prepare_ui_engine.py before configuring")
endif()
file(GLOB_RECURSE ENGINE_SOURCES CONFIGURE_DEPENDS ${ENGINE}/src/*.c)
add_library(lvgl STATIC ${ENGINE_SOURCES})
target_include_directories(lvgl PUBLIC ${ENGINE} ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE)
if(NOT CMAKE_CROSSCOMPILING)
  target_link_libraries(lvgl PUBLIC m)
endif()
add_library(avant_ui STATIC interface.cpp ../src/lyrics_model.cpp)
target_include_directories(avant_ui PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../vendor)
target_include_directories(avant_ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../src)
target_link_libraries(avant_ui PUBLIC lvgl)
target_compile_options(avant_ui PRIVATE -Wall -Wextra -Werror)
if(NOT CMAKE_CROSSCOMPILING)
  add_executable(ui-preview preview.cpp)
  target_link_libraries(ui-preview PRIVATE avant_ui)
else()
  add_library(ps4_ui_entry OBJECT ps4_main.cpp ps4_video.cpp)
  target_include_directories(ps4_ui_entry PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../build)
  target_link_libraries(ps4_ui_entry PRIVATE avant_ui)
endif()
